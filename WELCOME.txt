━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

    Create: src/hello.asm

    INCLUDE Irvine32.inc

    .data
        greeting BYTE "Hello from Assembly!",0

    .code
    main PROC
        mov  edx, OFFSET greeting
        call WriteString
        call Crlf
        exit
    main ENDP
    END main

    Build: .\build.bat src\hello.asm
    Run:   .\bin\hello.exe
